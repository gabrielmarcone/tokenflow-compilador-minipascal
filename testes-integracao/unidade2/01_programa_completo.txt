/* Teste 01 - programa correto que usa toda a linguagem:
   variáveis dos quatro tipos, procedimentos com parâmetros por valor
   e por referência (var), recursão, if/else, while, blocos begin/end,
   todos os operadores e as conversões permitidas.
   Resultado esperado: OK */
program Completo;
var i, n, soma: integer;
    media, taxa: real;
    letra: char;
    nome, frase: string;
var contador: integer;

procedure linha;
begin
  writeln("----------");
end;

procedure acumular(valor: integer; var total: integer);
begin
  total := total + valor;
end;

procedure fatorial(k: integer; var f: integer);
var anterior: integer;
begin
  if k <= 1 then
    f := 1
  else
  begin
    fatorial(k - 1, anterior);
    f := k * anterior;
  end;
end;

procedure saudar(quem: string; inicial: char; peso: real);
begin
  write("Olá, ", quem, " (", inicial, ") ", peso);
  writeln;
end;

begin
  read(n, taxa);
  soma := 0;
  i := 1;
  while i <= n do
  begin
    acumular(i * i, soma);
    i := i + 1;
  end;
  media := soma;
  media := (media + 0.5) * taxa - 1.5E2;
  contador := -n + (soma div 3) mod 7;
  fatorial(5, contador);
  letra := 'T';
  nome := letra;
  frase := nome + " Flow" + '!';
  if (soma > 10) = (contador <> 0) then
    if frase >= "A" then
      linha
    else
      writeln("antes de A");
  if letra < 'Z' then saudar(frase, letra, 2) else saudar("x", 'y', media);
  IF n = 0 THEN Writeln("vazio");
  while (i >= 1) do i := i - 1;
  begin
    LINHA;
    write(+media, -taxa, Soma);
  end;
end.
