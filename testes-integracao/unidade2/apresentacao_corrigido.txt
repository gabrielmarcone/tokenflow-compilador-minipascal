/* Demonstração do TokenFlow: a média de duas notas de um aluno.
   Este é apresentacao.txt com os quatro erros corrigidos (linhas 7, 16,
   21 e 23); as outras linhas são iguais. */
program Notas;
var nota1, nota2, media: real;
    faltas: integer;
    nome, situacao: string;

procedure calcular(a, b: real; var resultado: real);
begin
  resultado := (a + b) * 0.5;
end;

begin
  read(nome, nota1, nota2, faltas);
  calcular(nota1, nota2, media);
  if media >= 7.0 then
    situacao := "aprovado"
  else
    writeln(nome, " vai para a prova final");
  if faltas <> 0 then
    writeln("faltas: ", faltas);
  if faltas > 18 then writeln("reprovado por faltas");
end.
